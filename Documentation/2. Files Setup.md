Clone The Repo

`sudo apt install git -y`

`cd ~`

`git clone https://github.com/rishabhxchoudhary/taskTracker.git`

`cd taskTracker/backend/`

`touch .env`

`nano .env`

Add environment variables

`go build -o backend main.go`

`mkdir -p /home/ubuntu/taskTracker/backend`

`sudo mv /root/taskTracker/backend/backend /home/ubuntu/taskTracker/backend/`
`sudo chown -R ubuntu:ubuntu /home/ubuntu/taskTracker`
`chmod +x /home/ubuntu/taskTracker/backend/backend`

`sudo nano /etc/systemd/system/go-backend.service`

```
[Unit]
Description=Go Backend Service
After=network.target

[Service]
Type=simple
User=ubuntu
WorkingDirectory=/home/ubuntu/taskTracker/backend
ExecStart=/home/ubuntu/taskTracker/backend/backend
Restart=on-failure
Environment=GOPATH=/home/ubuntu/go
Environment=PATH=/usr/local/go/bin:/usr/bin:/bin

[Install]
WantedBy=multi-user.target
```

`sudo systemctl daemon-reload`

`sudo systemctl start go-backend`

`sudo systemctl enable go-backend`

`sudo systemctl status go-backend`
